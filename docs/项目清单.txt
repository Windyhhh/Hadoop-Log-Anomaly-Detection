   hdfs dfs -mkdir -p /user/hadoop/logs/input
   hdfs dfs -put 150/data/processed_logs.csv /user/hadoop/logs/input/

3. 运行MapReduce
   hadoop jar 150/mapreduce/target/log-anomaly-detector-1.0-SNAPSHOT.jar \

     /user/hadoop/logs/input \
